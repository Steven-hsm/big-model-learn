```python
# src/core/document_parser.py
"""文档解析器 - 支持PDF/Word/TXT/Markdown"""
from pathlib import Path
from typing import List, Dict
import logging

logger = logging.getLogger(__name__)

class DocumentParser:
    """统一文档解析接口"""

    def __init__(self):
        self.parsers = {
            '.pdf': self._parse_pdf,
            '.docx': self._parse_docx,
            '.txt': self._parse_txt,
            '.md': self._parse_txt,
        }

    def parse(self, file_path: str) -> Dict:
        """解析文档"""
        path = Path(file_path)
        ext = path.suffix.lower()

        if ext not in self.parsers:
            raise ValueError(f"不支持的文件格式: {ext}")

        parser = self.parsers[ext]
        content = parser(file_path)

        return {
            'filename': path.name,
            'file_type': ext,
            'content': content,
            'metadata': {
                'size': path.stat().st_size,
                'pages': content.count('\n\n') + 1,
            }
        }

    def _parse_pdf(self, file_path: str) -> str:
        """解析PDF"""
        import fitz  # PyMuPDF
        doc = fitz.open(file_path)
        text = ""
        for page_num, page in enumerate(doc):
            page_text = page.get_text()
            text += f"\n--- 第{page_num + 1}页 ---\n{page_text}"
        doc.close()
        return text

    def _parse_docx(self, file_path: str) -> str:
        """解析Word"""
        from docx import Document
        doc = Document(file_path)
        return "\n".join(para.text for para in doc.paragraphs if para.text.strip())

    def _parse_txt(self, file_path: str) -> str:
        """解析TXT/Markdown"""
        with open(file_path, 'r', encoding='utf-8') as f:
            return f.read()
```

```python
"""文本切分器"""
from typing import List, Dict
import re

class TextSplitter:
    """文本切分"""

    def __init__(self, chunk_size: int = 512, chunk_overlap: int = 50):
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap

    def split(self, text: str, metadata: Dict = None) -> List[Dict]:
        """切分文本为chunk"""
        # 先按段落分割
        paragraphs = re.split(r'\n\s*\n', text)

        chunks = []
        current_chunk = ""

        for para in paragraphs:
            para = para.strip()
            if not para:
                continue

            if len(current_chunk) + len(para) <= self.chunk_size:
                current_chunk += para + "\n\n"
            else:
                if current_chunk:
                    chunks.append({
                        'content': current_chunk.strip(),
                        'metadata': metadata or {},
                    })
                current_chunk = para + "\n\n"

        if current_chunk:
            chunks.append({
                'content': current_chunk.strip(),
                'metadata': metadata or {},
            })

        return chunks
```

```python
"""Embedding服务"""
from sentence_transformers import SentenceTransformer
from typing import List
import numpy as np
import logging

logger = logging.getLogger(__name__)

class EmbeddingService:
    """Embedding向量生成"""

    def __init__(self, model_name: str = "BAAI/bge-large-zh-v1.5"):
        logger.info(f"加载Embedding模型: {model_name}")
        self.model = SentenceTransformer(model_name)
        self.dimension = self.model.get_sentence_embedding_dimension()
        logger.info(f"Embedding维度: {self.dimension}")

    def encode(self, texts: List[str], batch_size: int = 32) -> np.ndarray:
        """批量生成Embedding"""
        embeddings = self.model.encode(
            texts,
            batch_size=batch_size,
            show_progress_bar=len(texts) > 100,
            normalize_embeddings=True,  # L2归一化，用于余弦相似度
        )
        return embeddings

    def encode_query(self, query: str) -> np.ndarray:
        """为查询生成Embedding（BGE模型需要加前缀）"""
        return self.model.encode(
            [f"为这个句子生成表示以用于检索相关文章：{query}"],
            normalize_embeddings=True,
        )[0]
```

```python
# src/core/retriever.py
"""检索服务 - 支持向量检索和BM25混合检索"""
import chromadb
from typing import List, Dict, Optional
import numpy as np
import logging

from src.core.embedding import EmbeddingService

logger = logging.getLogger(__name__)

class Retriever:
    """混合检索器"""

    def __init__(self, embedding_service: EmbeddingService, persist_dir: str = "./data/vectorstore"):
        self.embedding_service = embedding_service
        self.client = chromadb.PersistentClient(path=persist_dir)
        self.collection = self.client.get_or_create_collection(
            name="documents",
            metadata={"hnsw:space": "cosine"},
        )

    def add_documents(self, chunks: List[Dict]):
        """添加文档chunks到向量库"""
        ids = [f"chunk_{i}" for i in range(len(chunks))]
        texts = [c['content'] for c in chunks]
        metadatas = [c.get('metadata', {}) for c in chunks]

        embeddings = self.embedding_service.encode(texts)

        self.collection.add(
            ids=ids,
            embeddings=embeddings.tolist(),
            documents=texts,
            metadatas=metadatas,
        )
        logger.info(f"已添加 {len(chunks)} 个chunks到向量库")

    def search(self, query: str, top_k: int = 5) -> List[Dict]:
        """向量检索"""
        query_embedding = self.embedding_service.encode_query(query)

        results = self.collection.query(
            query_embeddings=[query_embedding.tolist()],
            n_results=top_k,
            include=["documents", "metadatas", "distances"],
        )

        search_results = []
        for i in range(len(results['ids'][0])):
            search_results.append({
                'content': results['documents'][0][i],
                'metadata': results['metadatas'][0][i],
                'score': 1 - results['distances'][0][i],  # 转换为相似度
            })

        return search_results
```

```python
# src/core/generator.py
"""生成服务 - LLM调用与回答生成"""
from openai import OpenAI
from typing import List, Dict, Optional
import logging

from src.config import settings

logger = logging.getLogger(__name__)

class Generator:
    """LLM生成服务"""

    def __init__(self):
        self.client = OpenAI(
            api_key=settings.LLM_API_KEY,
            base_url=settings.LLM_BASE_URL,
        )
        self.model = settings.LLM_MODEL

    def generate(
        self,
        query: str,
        contexts: List[Dict],
        conversation_history: Optional[List[Dict]] = None,
    ) -> Dict:
        """生成回答"""
        # 构建系统提示
        system_prompt = """你是一个专业的知识库助手。请根据提供的参考资料回答用户的问题。

要求：
1. 只根据提供的参考资料回答，不要编造信息
2. 如果参考资料中没有相关信息，请明确说明
3. 在回答中标注引用来源（用[1], [2]等标注）
4. 回答要简洁、准确、有条理"""

        # 构建上下文
        context_text = ""
        sources = []
        for i, ctx in enumerate(contexts):
            context_text += f"\n[参考资料{i+1}]\n{ctx['content']}\n"
            sources.append({
                'index': i + 1,
                'content': ctx['content'][:200],
                'score': ctx.get('score', 0),
                'metadata': ctx.get('metadata', {}),
            })

        # 构建消息列表
        messages = [{"role": "system", "content": system_prompt}]

        # 添加对话历史
        if conversation_history:
            messages.extend(conversation_history[-6:])  # 最近3轮

        # 添加当前查询
        user_message = f"参考资料：{context_text}\n\n用户问题：{query}"
        messages.append({"role": "user", "content": user_message})

        # 调用LLM
        response = self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            temperature=0.1,
            max_tokens=1024,
        )

        answer = response.choices[0].message.content

        return {
            'answer': answer,
            'sources': sources,
            'model': self.model,
            'usage': {
                'prompt_tokens': response.usage.prompt_tokens,
                'completion_tokens': response.usage.completion_tokens,
            }
        }
```

```python
"""问答查询API"""
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from typing import List, Optional
import logging

from src.core.retriever import Retriever
from src.core.generator import Generator
from src.core.embedding import EmbeddingService
from src.config import settings

logger = logging.getLogger(__name__)
router = APIRouter()

# 初始化服务（实际项目中应使用依赖注入）
embedding_service = EmbeddingService(settings.EMBEDDING_MODEL)
retriever = Retriever(embedding_service, settings.VECTORSTORE_PATH)
generator = Generator()

class QueryRequest(BaseModel):
    question: str
    top_k: int = 5
    conversation_id: Optional[str] = None

class SourceInfo(BaseModel):
    index: int
    content: str
    score: float

class QueryResponse(BaseModel):
    answer: str
    sources: List[SourceInfo]
    model: str

@router.post("/", response_model=QueryResponse)
async def query_rag(request: QueryRequest):
    """RAG问答查询"""
    try:
        # 1. 检索相关文档
        contexts = retriever.search(request.question, top_k=request.top_k)

        if not contexts:
            return QueryResponse(
                answer="抱歉，知识库中没有找到相关信息。",
                sources=[],
                model=settings.LLM_MODEL,
            )

        # 2. 生成回答
        result = generator.generate(
            query=request.question,
            contexts=contexts,
        )

        return QueryResponse(
            answer=result['answer'],
            sources=[SourceInfo(**s) for s in result['sources']],
            model=result['model'],
        )

    except Exception as e:
        logger.error(f"查询失败: {e}")
        raise HTTPException(status_code=500, detail=str(e))
```

```python
# src/core/document_parser.py - 重构为工厂模式
from abc import ABC, abstractmethod
from pathlib import Path
from typing import Dict, Optional
import logging

logger = logging.getLogger(__name__)

class BaseParser(ABC):
    """解析器基类"""

    @abstractmethod
    def parse(self, file_path: str) -> Dict:
        """解析文档，返回统一格式"""
        pass

    def _build_result(self, file_path: str, content: str, **extra) -> Dict:
        path = Path(file_path)
        return {
            'filename': path.name,
            'file_type': path.suffix.lower(),
            'content': content,
            'metadata': {
                'size': path.stat().st_size,
                **extra,
            }
        }

class PDFParser(BaseParser):
    """PDF解析器"""

    def parse(self, file_path: str) -> Dict:
        import fitz
        doc = fitz.open(file_path)
        pages = []
        for i, page in enumerate(doc):
            pages.append({
                'page_num': i + 1,
                'text': page.get_text(),
                # 提取表格
                'tables': self._extract_tables(page),
            })
        doc.close()
        full_text = "\n".join(f"--- 第{p['page_num']}页 ---\n{p['text']}" for p in pages)
        return self._build_result(file_path, full_text, pages=len(pages))

    def _extract_tables(self, page) -> list:
        """提取页面中的表格"""
        tables = page.find_tables()
        result = []
        for table in tables:
            df = table.to_pandas()
            if not df.empty:
                result.append(df.to_markdown())
        return result

class WordParser(BaseParser):
    """Word文档解析器"""

    def parse(self, file_path: str) -> Dict:
        from docx import Document
        doc = Document(file_path)

        sections = []
        for para in doc.paragraphs:
            if para.text.strip():
                sections.append(para.text)

        # 提取表格
        for table in doc.tables:
            rows = []
            for row in table.rows:
                rows.append([cell.text for cell in row.cells])
            if rows:
                sections.append("\n".join("\t".join(r) for r in rows))

        return self._build_result(file_path, "\n\n".join(sections))

class ExcelParser(BaseParser):
    """Excel解析器"""

    def parse(self, file_path: str) -> Dict:
        import pandas as pd
        xls = pd.ExcelFile(file_path)
        sheets = {}
        for name in xls.sheet_names:
            df = pd.read_excel(xls, sheet_name=name)
            sheets[name] = df.to_markdown()
        content = "\n\n".join(f"## {name}\n{md}" for name, md in sheets.items())
        return self._build_result(file_path, content, sheets=list(sheets.keys()))

class MarkdownParser(BaseParser):
    """Markdown解析器 - 保留标题结构"""

    def parse(self, file_path: str) -> Dict:
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
        return self._build_result(file_path, content)

class ParserFactory:
    """解析器工厂"""

    _parsers = {
        '.pdf': PDFParser,
        '.docx': WordParser,
        '.doc': WordParser,
        '.xlsx': ExcelParser,
        '.xls': ExcelParser,
        '.txt': MarkdownParser,
        '.md': MarkdownParser,
    }

    @classmethod
    def get_parser(cls, file_path: str) -> BaseParser:
        ext = Path(file_path).suffix.lower()
        parser_class = cls._parsers.get(ext)
        if not parser_class:
            raise ValueError(f"不支持的文件格式: {ext}，支持: {list(cls._parsers.keys())}")
        return parser_class()

    @classmethod
    def parse(cls, file_path: str) -> Dict:
        parser = cls.get_parser(file_path)
        return parser.parse(file_path)
```

```python
# src/core/text_splitter.py - 增强版
import re
from typing import List, Dict
from dataclasses import dataclass

@dataclass
class Chunk:
    content: str
    metadata: Dict
    token_count: int = 0

class MarkdownHeaderSplitter:
    """按Markdown标题切分"""

    def split(self, text: str, base_metadata: Dict = None) -> List[Chunk]:
        chunks = []
        current_headers = {}
        current_content = []

        for line in text.split('\n'):
            header_match = re.match(r'^(#{1,6})\s+(.+)$', line)

            if header_match:
                # 保存当前chunk
                if current_content:
                    content = '\n'.join(current_content).strip()
                    if content:
                        chunks.append(Chunk(
                            content=content,
                            metadata={**(base_metadata or {}), **current_headers},
                        ))

                # 更新标题层级
                level = len(header_match.group(1))
                current_headers[f'h{level}'] = header_match.group(2)
                current_content = [line]
            else:
                current_content.append(line)

        # 最后一个chunk
        if current_content:
            content = '\n'.join(current_content).strip()
            if content:
                chunks.append(Chunk(
                    content=content,
                    metadata={**(base_metadata or {}), **current_headers},
                ))

        return chunks

class SemanticSplitter:
    """语义切分 - 按语义边界切分"""

    def __init__(self, embedding_service, max_chunk_size: int = 512):
        self.embedding_service = embedding_service
        self.max_chunk_size = max_chunk_size

    def split(self, text: str, base_metadata: Dict = None) -> List[Chunk]:
        """先按段落切，然后合并语义相似的段落"""
        paragraphs = [p.strip() for p in text.split('\n\n') if p.strip()]

        if len(paragraphs) <= 1:
            return [Chunk(content=text, metadata=base_metadata or {})]

        # 生成段落embedding
        embeddings = self.embedding_service.encode(paragraphs)

        # 计算相邻段落的相似度
        from numpy import dot
        similarities = []
        for i in range(len(embeddings) - 1):
            sim = dot(embeddings[i], embeddings[i + 1])
            similarities.append(sim)

        # 在相似度低谷处切分
        threshold = sum(similarities) / len(similarities) - 0.5 * self._std(similarities)

        chunks = []
        current_text = paragraphs[0]

        for i, sim in enumerate(similarities):
            if sim < threshold or len(current_text) > self.max_chunk_size:
                chunks.append(Chunk(
                    content=current_text,
                    metadata=base_metadata or {},
                ))
                current_text = paragraphs[i + 1]
            else:
                current_text += '\n\n' + paragraphs[i + 1]

        if current_text:
            chunks.append(Chunk(
                content=current_text,
                metadata=base_metadata or {},
            ))

        return chunks

    @staticmethod
    def _std(values):
        import numpy as np
        return float(np.std(values))
```

```python
# src/core/retriever.py - 增强版：混合检索
import chromadb
from typing import List, Dict, Optional
import numpy as np
import logging
import math
from collections import Counter

from src.core.embedding import EmbeddingService

logger = logging.getLogger(__name__)

class BM25Retriever:
    """BM25文本检索"""

    def __init__(self, k1: float = 1.5, b: float = 0.75):
        self.k1 = k1
        self.b = b
        self.corpus = []
        self.doc_freqs = []
        self.idf = {}
        self.avgdl = 0

    def fit(self, documents: List[str]):
        """构建BM25索引"""
        self.corpus = [doc.lower().split() for doc in documents]
        n_docs = len(self.corpus)
        self.avgdl = sum(len(doc) for doc in self.corpus) / max(n_docs, 1)

        # 计算IDF
        df = Counter()
        for doc in self.corpus:
            df.update(set(doc))

        for term, freq in df.items():
            self.idf[term] = math.log((n_docs - freq + 0.5) / (freq + 0.5) + 1)

    def search(self, query: str, top_k: int = 10) -> List[Dict]:
        """BM25检索"""
        query_terms = query.lower().split()
        scores = []

        for doc_idx, doc in enumerate(self.corpus):
            score = 0
            doc_len = len(doc)
            term_freqs = Counter(doc)

            for term in query_terms:
                if term in self.idf:
                    tf = term_freqs.get(term, 0)
                    idf = self.idf[term]
                    numerator = tf * (self.k1 + 1)
                    denominator = tf + self.k1 * (1 - self.b + self.b * doc_len / self.avgdl)
                    score += idf * numerator / denominator

            scores.append(score)

        # 排序
        ranked = sorted(enumerate(scores), key=lambda x: x[1], reverse=True)[:top_k]
        return [{'index': idx, 'score': score} for idx, score in ranked]

class HybridRetriever:
    """混合检索：向量 + BM25 + 融合排序"""

    def __init__(self, embedding_service: EmbeddingService, persist_dir: str = "./data/vectorstore"):
        self.embedding_service = embedding_service
        self.client = chromadb.PersistentClient(path=persist_dir)
        self.collection = self.client.get_or_create_collection(
            name="documents",
            metadata={"hnsw:space": "cosine"},
        )
        self.bm25 = BM25Retriever()
        self._bm25_fitted = False

    def add_documents(self, chunks: List[Dict]):
        """添加文档"""
        ids = [f"chunk_{i}_{hash(c['content']) % 100000}" for i, c in enumerate(chunks)]
        texts = [c['content'] for c in chunks]
        metadatas = [c.get('metadata', {}) for c in chunks]

        # 向量索引
        embeddings = self.embedding_service.encode(texts)
        self.collection.add(
            ids=ids,
            embeddings=embeddings.tolist(),
            documents=texts,
            metadatas=metadatas,
        )

        # BM25索引
        self.bm25.fit(texts)
        self._bm25_fitted = True
        self._all_chunks = chunks

        logger.info(f"已索引 {len(chunks)} 个chunks (向量+BM25)")

    def search(self, query: str, top_k: int = 5, alpha: float = 0.7) -> List[Dict]:
        """混合检索 + 融合排序"""
        # 向量检索
        query_embedding = self.embedding_service.encode_query(query)
        vector_results = self.collection.query(
            query_embeddings=[query_embedding.tolist()],
            n_results=top_k * 3,  # 多检索一些用于融合
            include=["documents", "metadatas", "distances"],
        )

        vector_scores = {}
        for i in range(len(vector_results['ids'][0])):
            doc_id = vector_results['ids'][0][i]
            similarity = 1 - vector_results['distances'][0][i]
            vector_scores[doc_id] = {
                'content': vector_results['documents'][0][i],
                'metadata': vector_results['metadatas'][0][i],
                'vector_score': similarity,
            }

        # BM25检索
        if self._bm25_fitted:
            bm25_results = self.bm25.search(query, top_k=top_k * 3)
            for result in bm25_results:
                idx = result['index']
                if idx < len(self._all_chunks):
                    content = self._all_chunks[idx]['content']
                    doc_id = f"bm25_{idx}"
                    if doc_id not in vector_scores:
                        vector_scores[doc_id] = {
                            'content': content,
                            'metadata': self._all_chunks[idx].get('metadata', {}),
                            'vector_score': 0,
                        }
                    vector_scores[doc_id]['bm25_score'] = result['score']

        # 融合排序（Reciprocal Rank Fusion）
        return self._reciprocal_rank_fusion(vector_scores, top_k, alpha)

    def _reciprocal_rank_fusion(self, scores: Dict, top_k: int, alpha: float, k: int = 60) -> List[Dict]:
        """RRF融合排序"""
        # 向量排序
        vector_ranked = sorted(
            scores.items(),
            key=lambda x: x[1].get('vector_score', 0),
            reverse=True,
        )

        # BM25排序
        bm25_ranked = sorted(
            scores.items(),
            key=lambda x: x[1].get('bm25_score', 0),
            reverse=True,
        )

        # 计算RRF分数
        rrf_scores = {}
        for rank, (doc_id, _) in enumerate(vector_ranked):
            rrf_scores[doc_id] = rrf_scores.get(doc_id, 0) + alpha / (k + rank + 1)

        for rank, (doc_id, _) in enumerate(bm25_ranked):
            rrf_scores[doc_id] = rrf_scores.get(doc_id, 0) + (1 - alpha) / (k + rank + 1)

        # 最终排序
        final_ranked = sorted(rrf_scores.items(), key=lambda x: x[1], reverse=True)[:top_k]

        results = []
        for doc_id, rrf_score in final_ranked:
            info = scores[doc_id]
            results.append({
                'content': info['content'],
                'metadata': info.get('metadata', {}),
                'score': rrf_score,
            })

        return results
```

```python
from sentence_transformers import CrossEncoder
from typing import List, Dict
import logging

logger = logging.getLogger(__name__)

class Reranker:
    """使用Cross-Encoder进行精排"""

    def __init__(self, model_name: str = "BAAI/bge-reranker-large"):
        logger.info(f"加载Reranker模型: {model_name}")
        self.model = CrossEncoder(model_name)

    def rerank(
        self,
        query: str,
        documents: List[Dict],
        top_k: int = 5,
    ) -> List[Dict]:
        """对检索结果进行精排"""
        if not documents:
            return []

        # 构建query-document对
        pairs = [(query, doc['content']) for doc in documents]

        # 计算相关性分数
        scores = self.model.predict(pairs)

        # 按分数排序
        scored_docs = list(zip(documents, scores))
        scored_docs.sort(key=lambda x: x[1], reverse=True)

        # 返回top_k
        results = []
        for doc, score in scored_docs[:top_k]:
            result = doc.copy()
            result['rerank_score'] = float(score)
            results.append(result)

        logger.info(f"Rerank完成: {len(documents)} → {len(results)}")
        return results
```

```python
from openai import OpenAI

要求：
1. 只根据提供的参考资料回答，不要编造信息

        for i, ctx in enumerate(contexts):
            metadata = ctx.get('metadata', {})
            page_info = ""
            if 'page_num' in metadata:
                page_info = f" (第{metadata['page_num']}页)"
            elif 'h1' in metadata:
                page_info = f" (章节: {metadata['h1']})"

            context_parts.append(f"[参考资料{i+1}]{page_info}\n{ctx['content']}")
            sources.append({
                'index': i + 1,
                'content': ctx['content'][:300],
                'page': metadata.get('page_num'),
                'section': metadata.get('h1', metadata.get('h2', '')),
                'score': ctx.get('score', 0),
                'rerank_score': ctx.get('rerank_score'),
            })

        context_text = "\n\n".join(context_parts)

        # 构建消息
        messages = [{"role": "system", "content": system_prompt}]
        if conversation_history:
            messages.extend(conversation_history[-6:])
        messages.append({
            "role": "user",
            "content": f"参考资料：\n{context_text}\n\n用户问题：{query}"
        })

        response = self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            temperature=0.1,
            max_tokens=1024,
        )

        answer = response.choices[0].message.content

        # 提取回答中实际引用的编号
        cited_indices = set(int(x) for x in re.findall(r'\[(\d+)\]', answer))
        cited_sources = [s for s in sources if s['index'] in cited_indices]

        return {
            'answer': answer,
            'sources': cited_sources if cited_sources else sources[:3],
            'model': self.model,
            'usage': {
                'prompt_tokens': response.usage.prompt_tokens,
                'completion_tokens': response.usage.completion_tokens,
            }
        }
```

```python
    token = credentials.credentials
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        user_id: str = payload.get("sub")
        role: str = payload.get("role")

            raise HTTPException(status_code=401, detail="Invalid token")
        return {"user_id": user_id, "role": role}
    except JWTError:
        raise HTTPException(status_code=401, detail="Invalid token")

def require_permission(permission: str):
    def permission_checker(current_user: dict = Depends(get_current_user)):
        user_perms = ROLE_PERMISSIONS.get(current_user["role"], [])
```

```python
                ground_truth=q.ground_truth,
                sources=gen_result['sources'],
                retrieval_recall=retrieval_recall,
                retrieval_mrr=retrieval_mrr,
                faithfulness=faithfulness,
                relevancy=relevancy,
            ))

        # 汇总指标
        avg_recall = sum(r.retrieval_recall for r in results) / len(results)
        avg_mrr = sum(r.retrieval_mrr for r in results) / len(results)
        avg_faithfulness = sum(r.faithfulness for r in results) / len(results)
        avg_relevancy = sum(r.relevancy for r in results) / len(results)

        report = {
            'total_questions': len(results),
            'avg_retrieval_recall': avg_recall,
            'avg_retrieval_mrr': avg_mrr,
            'avg_faithfulness': avg_faithfulness,
            'avg_relevancy': avg_relevancy,
            'details': results,
        }

        logger.info(f"评估完成: Recall={avg_recall:.3f}, MRR={avg_mrr:.3f}, "
                     f"Faithfulness={avg_faithfulness:.3f}, Relevancy={avg_relevancy:.3f}")

        return report

    def _calc_recall(self, contexts: List[Dict], relevant_ids: List[str], k: int = 5) -> float:
        """Recall@K: 前K个检索结果中包含多少相关文档"""
        if not relevant_ids:
            return 1.0
        retrieved_ids = set()
        for ctx in contexts[:k]:
            content_hash = str(hash(ctx['content']) % 100000)
            retrieved_ids.add(content_hash)
        relevant_set = set(relevant_ids)
        return len(retrieved_ids & relevant_set) / len(relevant_set)

    def _calc_mrr(self, contexts: List[Dict], relevant_ids: List[str]) -> float:
        """MRR: 第一个相关文档的排名倒数"""
        if not relevant_ids:
            return 1.0
        for rank, ctx in enumerate(contexts, 1):
            content_hash = str(hash(ctx['content']) % 100000)
            if content_hash in relevant_ids:
                return 1.0 / rank
        return 0.0

    def _eval_faithfulness(self, answer: str, contexts: List[Dict]) -> float:
        """Faithfulness评估（LLM-as-Judge）：回答是否忠于上下文"""
        context_text = "\n".join(c['content'] for c in contexts)

        prompt = f"""请评估以下回答是否完全基于给定的上下文信息，没有编造内容。

上下文：{context_text}

回答：{answer}

评分标准：
1.0 - 完全基于上下文，没有编造
0.5 - 大部分基于上下文，有少量编造
0.0 - 大量编造内容

请只返回一个分数(0.0/0.5/1.0)："""

        # 实际项目中调用LLM评分
        # 这里简化为返回0.8作为示例
        return 0.8

    def _eval_relevancy(self, question: str, answer: str) -> float:
        """Relevancy评估（LLM-as-Judge）：回答是否切题"""
        prompt = f"""请评估以下回答是否准确回答了用户的问题。

问题：{question}
回答：{answer}

评分标准：
1.0 - 完全切题，准确回答了问题
0.5 - 部分切题，但不够完整或偏题
0.0 - 完全不切题

请只返回一个分数(0.0/0.5/1.0)："""

        return 0.8
```

```python
"""缓存层：Embedding缓存 + LLM响应缓存"""
import redis
import hashlib
import json
import numpy as np
from typing import Optional, List, Dict
import logging

from src.config import settings

logger = logging.getLogger(__name__)

class EmbeddingCache:
    """Embedding缓存 - 避免重复计算"""

    def __init__(self, redis_url: str = None):
        try:
            self.redis = redis.from_url(redis_url or settings.REDIS_URL)
            self.redis.ping()
            self.enabled = True
        except:
            logger.warning("Redis不可用，Embedding缓存禁用")
            self.enabled = False
            self._memory_cache = {}

    def _cache_key(self, text: str) -> str:
        return f"emb:{hashlib.md5(text.encode()).hexdigest()}"

    def get(self, text: str) -> Optional[np.ndarray]:
        key = self._cache_key(text)
        if self.enabled:
            cached = self.redis.get(key)
            if cached:
                return np.frombuffer(cached, dtype=np.float32)
        else:
            if key in self._memory_cache:
                return self._memory_cache[key]
        return None

    def put(self, text: str, embedding: np.ndarray):
        key = self._cache_key(text)
        if self.enabled:
            self.redis.setex(key, 86400 * 7, embedding.astype(np.float32).tobytes())  # 7天过期
        else:
            self._memory_cache[key] = embedding

class LLMResponseCache:
    """LLM响应缓存 - 相同问题返回缓存结果"""

    def __init__(self, redis_url: str = None):
        try:
            self.redis = redis.from_url(redis_url or settings.REDIS_URL)
            self.enabled = True
        except:
            self.enabled = False
            self._memory_cache = {}

    def _cache_key(self, query: str, context_hash: str) -> str:
        return f"llm:{hashlib.md5(f'{query}:{context_hash}'.encode()).hexdigest()}"

    def get(self, query: str, contexts: List[Dict]) -> Optional[Dict]:
        context_hash = hashlib.md5(
            json.dumps([c['content'] for c in contexts], sort_keys=True).encode()
        ).hexdigest()
        key = self._cache_key(query, context_hash)

        if self.enabled:
            cached = self.redis.get(key)
            if cached:
                logger.info("LLM缓存命中")
                return json.loads(cached)
        else:
            if key in self._memory_cache:
                return self._memory_cache[key]
        return None

    def put(self, query: str, contexts: List[Dict], response: Dict, ttl: int = 3600):
        context_hash = hashlib.md5(
            json.dumps([c['content'] for c in contexts], sort_keys=True).encode()
        ).hexdigest()
        key = self._cache_key(query, context_hash)

        if self.enabled:
            self.redis.setex(key, ttl, json.dumps(response, ensure_ascii=False))
        else:
            self._memory_cache[key] = response
```

```python
# src/api/routes/query.py - 增强版：异步 + 流式输出
from fastapi import APIRouter, HTTPException
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
from typing import List, Optional
import asyncio
import logging

from src.core.retriever import HybridRetriever
from src.core.reranker import Reranker
from src.core.generator import Generator
from src.core.embedding import EmbeddingService
from src.core.cache import LLMResponseCache
from src.core.conversation import ConversationManager
from src.config import settings

logger = logging.getLogger(__name__)
router = APIRouter()

# 初始化服务
embedding_service = EmbeddingService(settings.EMBEDDING_MODEL)
retriever = HybridRetriever(embedding_service, settings.VECTORSTORE_PATH)
reranker = Reranker()
generator = Generator()
llm_cache = LLMResponseCache()
conv_manager = ConversationManager()

class QueryRequest(BaseModel):
    question: str
    top_k: int = 5
    conversation_id: Optional[str] = None
    stream: bool = False

class QueryResponse(BaseModel):
    answer: str
    sources: List[dict]
    model: str
    conversation_id: str

@router.post("/", response_model=QueryResponse)
async def query_rag(request: QueryRequest):
    """RAG问答查询"""
    # 创建或获取对话
    conv_id = request.conversation_id or conv_manager.create_conversation()

    # 异步检索
    contexts = await asyncio.to_thread(
        retriever.search, request.question, request.top_k
    )

    # Reranker精排
    if contexts:
        contexts = reranker.rerank(request.question, contexts, top_k=3)

    # 检查缓存
    cached = llm_cache.get(request.question, contexts)
    if cached:
        return QueryResponse(**cached, conversation_id=conv_id)

    # 获取对话历史
    history = conv_manager.get_history(conv_id)

    # 异步生成
    result = await asyncio.to_thread(
        generator.generate,
        query=request.question,
        contexts=contexts,
        conversation_history=history,
    )

    # 保存对话
    conv_manager.add_message(conv_id, "user", request.question)
    conv_manager.add_message(conv_id, "assistant", result['answer'], result['sources'])

    # 缓存响应
    response = {
        'answer': result['answer'],
        'sources': result['sources'],
        'model': result['model'],
    }
    llm_cache.put(request.question, contexts, response)

    return QueryResponse(**response, conversation_id=conv_id)

@router.post("/stream")
async def query_stream(request: QueryRequest):
    """流式输出"""
    import json

    def generate():
        contexts = retriever.search(request.question, request.top_k)
        if contexts:
            contexts = reranker.rerank(request.question, contexts, top_k=3)

        # 先发送检索结果
        sources_data = json.dumps({"type": "sources", "data": [
            {"content": c['content'][:200], "score": c.get('score', 0)}
            for c in contexts
        ]}, ensure_ascii=False)
        yield f"data: {sources_data}\n\n"

        # 流式生成回答（简化示例）
        full_answer = ""
        for char in "这是一个流式回答的示例。实际项目中使用LLM的stream模式。":
            full_answer += char
            chunk = json.dumps({"type": "content", "data": char}, ensure_ascii=False)
            yield f"data: {chunk}\n\n"

        yield f"data: {json.dumps({'type': 'done'})}\n\n"

    return StreamingResponse(generate(), media_type="text/event-stream")
```

## 六、自测题

### 题目

1. **RAG系统中混合检索（向量+BM25）相比单一检索有什么优势？融合排序的RRF算法原理是什么？**

2. **Reranker精排的作用是什么？为什么不在检索阶段直接使用Cross-Encoder？**

3. **多轮对话中如何管理上下文窗口？ConversationBufferWindowMemory的优缺点是什么？**

4. **RAG评估体系中Faithfulness和Relevancy分别评估什么？如何使用LLM-as-Judge进行自动评估？**

5. **在大型项目中，如何保证代码质量和团队协作效率？列举至少3个关键实践。**

### 参考答案

<details>
<summary>点击展开答案</summary>

1. **混合检索优势**：向量检索擅长语义相似度匹配（"如何学习AI"能匹配"机器学习入门方法"），BM25擅长关键词精确匹配（专有名词、术语）。两者互补，召回率更高。**RRF (Reciprocal Rank Fusion) 原理**：对每种检索方法的结果按排名赋予分数 score = 1/(k+rank)，k通常为60。然后对每个文档在所有方法中的RRF分数求和作为最终分数。这样不同评分尺度的检索方法可以在同一框架下融合。

2. **Reranker作用**：对粗排结果进行精排，提升top-k结果的质量。Cross-Encoder同时编码query和document，能捕捉更细粒度的语义关系。**不在检索阶段使用Cross-Encoder的原因**：Cross-Encoder需要对每个query-document对分别计算，复杂度为O(N)，N为文档总数。而Bi-Encoder（向量检索）可以预先计算document embedding，查询时只需O(1)的向量相似度计算。因此先用Bi-Encoder粗排取top-N，再用Cross-Encoder精排取top-K。

3. **上下文管理**：ConversationBufferWindowMemory保留最近N轮对话作为LLM上下文。**优点**：实现简单，计算开销小，保持短期对话连贯性。**缺点**：(1) 固定窗口会丢失早期重要信息 (2) 窗口过小导致上下文不连贯 (3) 窗口过大浪费token。更好的方案：摘要记忆（对早期对话压缩为摘要）+ 关键信息提取（保存用户偏好等重要事实）。

4. **Faithfulness（忠实度）**：评估生成回答是否完全基于检索到的上下文，没有编造信息（幻觉）。**Relevancy（相关性）**：评估回答是否准确回答了用户的问题。**LLM-as-Judge**：使用强大的LLM（如GPT-4）作为评审，给定评分标准和上下文，让LLM对回答进行打分。优点是自动化程度高，缺点是存在评估偏差和成本问题。通常需要设计清晰的评分标准和示例。

5. **关键实践**：(1) **代码规范** - 使用black/isort/flake8统一代码风格，配置pre-commit hooks自动检查。(2) **Code Review** - 所有代码合并前必须经过Review，确保质量和知识共享。(3) **自动化测试** - pytest单元测试覆盖核心逻辑，CI自动运行测试。(4) **模块化设计** - 清晰的目录结构和接口定义，降低耦合度。(5) **文档驱动** - 先写接口文档再实现，确保设计合理。

## 七、Java开发者提示

> 帮助有Java背景的开发者理解综合项目中的工程概念

| 综合项目概念 | Java类比 | 说明 |
|------------|---------|------|

| Pydantic Model | Java Record/POJO | 数据验证和序列化 |
| 依赖注入(Depends) | Spring @Autowired | FastAPI的依赖注入机制 |
| 工厂模式(ParserFactory) | Java工厂模式 | 设计模式通用 |
| BM25检索 | Elasticsearch全文检索 | 全文搜索算法 |
| 向量检索 | Elasticsearch kNN Search | 语义检索 |
| Reranker | 搜索结果二次排序 | 精排概念相同 |
| JWT认证 | Spring Security JWT | 认证方案相同 |
| RBAC | Spring Security RBAC | 权限控制概念相同 |
| Redis缓存 | Spring Cache + Redis | 缓存方案相同 |
| Gradio前端 | Thymeleaf/Vaadin | 快速构建Web UI |
| pytest | JUnit | 测试框架类比 |
| locust | JMeter | 性能压测工具类比 |
| Docker Compose | Docker Compose(通用) | 容器编排方式相同 |
| GitHub Actions | Jenkins/GitHub Actions(通用) | CI/CD工具通用 |

### 关键思维转换

1. **Python更灵活**: Python没有Java那么严格的类型系统，但通过type hints和Pydantic可以实现类似的类型安全
2. **异步是常态**: FastAPI天然支持async/await，比Java的CompletableFuture更直观
3. **函数是一等公民**: Python中函数可以作为参数传递，无需Java的函数式接口
4. **快速原型优先**: Python生态鼓励快速验证想法，再逐步优化
5. **AI工程化 = 软件工程 + AI**: 核心还是软件工程能力，加上对AI模型和数据的理解
