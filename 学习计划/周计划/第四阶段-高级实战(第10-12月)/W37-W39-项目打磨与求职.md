```python
# 重构后：类型注解 + 工厂模式 + 错误处理 + 文档字符串
from pathlib import Path
from typing import Dict, Protocol
import logging

logger = logging.getLogger(__name__)

class DocumentParseError(Exception):
    """文档解析异常"""
    pass

class BaseParser(Protocol):
    """解析器接口"""
    def parse(self, file_path: Path) -> str: ...

class PDFParser:
    """PDF文档解析器"""

    def parse(self, file_path: Path) -> str:
        try:
            import fitz
            doc = fitz.open(str(file_path))
            pages = [page.get_text() for page in doc]
            doc.close()
            return "\n".join(pages)
        except Exception as e:
            raise DocumentParseError(f"PDF解析失败 {file_path}: {e}") from e

class WordParser:
    """Word文档解析器"""

    def parse(self, file_path: Path) -> str:
        try:
            from docx import Document
            doc = Document(str(file_path))
            return "\n".join(p.text for p in doc.paragraphs if p.text.strip())
        except Exception as e:
            raise DocumentParseError(f"Word解析失败 {file_path}: {e}") from e

class TextParser:
    """纯文本文档解析器"""

    def parse(self, file_path: Path) -> str:
        try:
            return file_path.read_text(encoding='utf-8')
        except Exception as e:
            raise DocumentParseError(f"文本读取失败 {file_path}: {e}") from e

# 工厂
PARSER_REGISTRY: Dict[str, type] = {
    '.pdf': PDFParser,
    '.docx': WordParser,
    '.doc': WordParser,
    '.txt': TextParser,
    '.md': TextParser,
}

def parse_document(file_path: str | Path) -> Dict[str, str | int | dict]:
    """解析文档，返回统一格式的结果。

    Args:
        file_path: 文档文件路径

    Returns:
        包含content、metadata等字段的字典

    Raises:
        DocumentParseError: 文档解析失败时抛出
        ValueError: 不支持的文件格式
    """
    path = Path(file_path)
    if not path.exists():
        raise DocumentParseError(f"文件不存在: {file_path}")

    parser_class = PARSER_REGISTRY.get(path.suffix.lower())
    if parser_class is None:
        supported = ", ".join(PARSER_REGISTRY.keys())
        raise ValueError(f"不支持的格式 '{path.suffix}'，支持: {supported}")

    parser = parser_class()
    content = parser.parse(path)
    logger.info(f"文档解析完成: {path.name} ({len(content)}字符)")

    return {
        'filename': path.name,
        'file_type': path.suffix.lower(),
        'content': content,
        'metadata': {
            'size_bytes': path.stat().st_size,
            'char_count': len(content),
        },
    }
```

```python
        times = []
        for _ in range(100):

            bm25.search("关键词", top_k=10)
```

### Day 1: AI基础复习 (2h)

```
ML核心知识点：

1. 线性回归
   - 损失函数：MSE = 1/n * Σ(y_i - ŷ_i)²
   - 解析解：w = (X^T X)^{-1} X^T y
   - 正则化：L1(Lasso) vs L2(Ridge)

2. 逻辑回归
   - Sigmoid函数：σ(z) = 1/(1+e^{-z})
   - 损失函数：交叉熵
   - 多分类：Softmax

3. 决策树与集成方法
```

```python
# 手写题1：实现Self-Attention
import torch
import torch.nn as nn
import torch.nn.functional as F
import math

class SelfAttention(nn.Module):
    """手写Self-Attention实现"""

    def __init__(self, d_model: int, n_heads: int):
        super().__init__()
        assert d_model % n_heads == 0

        self.d_model = d_model
        self.n_heads = n_heads
        self.d_k = d_model // n_heads

        self.W_q = nn.Linear(d_model, d_model)
        self.W_k = nn.Linear(d_model, d_model)
        self.W_v = nn.Linear(d_model, d_model)
        self.W_o = nn.Linear(d_model, d_model)

    def forward(self, x: torch.Tensor, mask: torch.Tensor = None) -> torch.Tensor:
        batch_size, seq_len, _ = x.shape

        # 线性变换
        Q = self.W_q(x)  # (batch, seq, d_model)
        K = self.W_k(x)
        V = self.W_v(x)

        # 分多头
        Q = Q.view(batch_size, seq_len, self.n_heads, self.d_k).transpose(1, 2)
        K = K.view(batch_size, seq_len, self.n_heads, self.d_k).transpose(1, 2)
        V = V.view(batch_size, seq_len, self.n_heads, self.d_k).transpose(1, 2)
        # (batch, n_heads, seq, d_k)

        # 缩放点积注意力
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.d_k)
        # (batch, n_heads, seq, seq)

        if mask is not None:
            scores = scores.masked_fill(mask == 0, float('-inf'))

        attn_weights = F.softmax(scores, dim=-1)
        context = torch.matmul(attn_weights, V)
        # (batch, n_heads, seq, d_k)

        # 合并多头
        context = context.transpose(1, 2).contiguous()
        context = context.view(batch_size, seq_len, self.d_model)

        # 输出投影
        output = self.W_o(context)
        return output
```

```python
# 手写题2：实现Focal Loss
import torch
import torch.nn as nn
import torch.nn.functional as F

class FocalLoss(nn.Module):
    """Focal Loss：解决类别不平衡问题

    FL(p_t) = -alpha_t * (1 - p_t)^gamma * log(p_t)

    alpha: 正负样本权重
    gamma: 聚焦参数，gamma越大越关注难样本
    """

    def __init__(self, alpha: float = 0.25, gamma: float = 2.0, reduction: str = 'mean'):
        super().__init__()
        self.alpha = alpha
        self.gamma = gamma
        self.reduction = reduction

    def forward(self, logits: torch.Tensor, targets: torch.Tensor) -> torch.Tensor:
        # 计算交叉熵（不减入）
        ce_loss = F.binary_cross_entropy_with_logits(logits, targets, reduction='none')

        # 计算p_t
        p_t = torch.exp(-ce_loss)

        # 计算alpha_t
        alpha_t = self.alpha * targets + (1 - self.alpha) * (1 - targets)

        # Focal Loss
        focal_loss = alpha_t * (1 - p_t) ** self.gamma * ce_loss

        if self.reduction == 'mean':
            return focal_loss.mean()
        elif self.reduction == 'sum':
            return focal_loss.sum()
        return focal_loss
```

### 参考答案

<details>
<summary>点击展开答案</summary>

1. **STAR** = **S**ituation（情境）+ **T**ask（任务）+ **A**ction（行动）+ **R**esult（结果）。示例："在开发RAG系统时(S)，我需要将检索准确率从71%提升到80%以上(T)。我对比了3种检索策略——纯向量检索、纯BM25、混合检索+RRF融合，最终选择了混合检索方案并集成了Reranker精排(A)。结果检索Recall@5从71%提升到89%，MRR从0.65提升到0.82，延迟P95控制在0.4s以内(R)。"

2. **设计RAG系统的维度**：(1) **需求澄清** - 用户规模、文档量、文档类型、延迟要求。(2) **文档处理Pipeline** - 解析→切分→Embedding的架构设计，关键决策：切分策略（固定/语义/混合）、Embedding模型选型。(3) **检索服务** - 向量数据库选型、检索策略（纯向量/混合）、Reranker集成。(4) **生成服务** - LLM选型（API/自部署）、Prompt工程、引用溯源。(5) **扩展性** - 缓存策略、异步处理、水平扩展。(6) **评估体系** - 检索指标（Recall/MRR）+ 生成指标（Faithfulness/Relevancy）。(7) **监控** - 延迟/QPS/数据漂移监控。

3. **量化对比**：(1) ❌ "优化了检索性能" → ✅ "检索延迟P95从2.8s降低到0.4s（降低85%）"。(2) ❌ "提高了准确率" → ✅ "检索Recall@5从71%提升到89%（提升25%）"。(3) ❌ "开发了一个RAG系统" → ✅ "开发了支持6种文档格式的RAG系统，索引10万+文档，支持50 QPS"。量化让成果可衡量、可比较，面试官能直观理解你的能力水平。

4. **面试复盘关注**：(1) **知识盲区** - 哪些问题答不上来或答不好，需要补充学习。(2) **表达方式** - 回答是否清晰有条理，STAR法则运用是否熟练。(3) **时间管理** - 是否在规定时间内完成了回答，哪些地方太啰嗦或太简略。(4) **面试官关注点** - 面试官追问了哪些方向，说明他们更看重什么能力。(5) **技术深度** - 是否有被问到很深入的技术细节，需要加强哪些知识点。(6) **行为面试** - 团队协作、技术决策等软技能问题的回答是否充分。利用方法：每次面试后立即记录，每周回顾总结，针对薄弱环节集中练习。

5. **差异化优势**：(1) **工程化能力** - Java经验带来的系统设计、架构设计、代码质量意识是很多纯AI背景候选人欠缺的。(2) **生产级思维** - 强调你的项目不是玩具项目，有完整的测试、监控、部署、CI/CD。(3) **全栈视角** - 能理解从数据处理到模型训练到服务部署到前端展示的全链路。(4) **MLOps优势** - Java开发者在CI/CD、微服务、监控等方面的基础可以直接迁移。(5) **学习速度** - 9个月系统学习并完成3个完整项目，证明学习能力和自驱力。(6) **实际经验** - 开源贡献、技术博客、性能数据都是可验证的能力证明。

## 十、Java开发者提示

> 帮助有Java背景的开发者准备AI岗位求职

| 求职概念 | Java经验迁移 | 说明 |
|---------|------------|------|
| 项目讲述 | Java项目讲述经验 | STAR法则通用 |
| 系统设计 | 微服务架构设计 | 设计思维通用，关注ML特有组件 |
| 编码面试 | LeetCode刷题经验 | 算法题通用，AI岗位还考ML手写题 |
| 简历优化 | Java简历优化经验 | 量化方法通用，突出AI项目 |
| 行为面试 | 行为面试(通用) | 回答框架相同 |
| 技术博客 | 技术影响力建设 | 博客写作经验通用 |

### 关键思维转换

1. **AI岗位更看重项目**: 相比Java岗位看重框架经验，AI岗位更看重你做过什么项目、解决过什么问题
2. **面试官可能关注不同层面**: 有的关注ML理论，有的关注工程实现，有的关注业务理解，需要全面准备
3. **开源贡献是加分项**: 在AI领域，GitHub上的项目比证书更有说服力
4. **工程+AI的复合能力是稀缺的**: 不要觉得Java经验是劣势，它是很多AI岗位急需的能力
5. **面试是双向选择**: 通过面试了解团队技术栈和文化，选择最适合自己的机会
